   701→function VideoPlayer({ src, title, subtitle }: VideoPlayerProps) {
   702→  const videoRef = useRef<HTMLVideoElement>(null)
   703→  const [isLoading, setIsLoading] = useState(true)
   704→  const [duration, setDuration] = useState(0)
   705→
   706→  useEffect(() => {
   707→    const video = videoRef.current
   708→    if (!video) return
   709→    const handleLoadedMetadata = () => {
   710→      setDuration(video.duration || 0)
   711→      setIsLoading(false)
   712→    }
   713→    const handleCanPlay = () => setIsLoading(false)
   714→    video.addEventListener('loadedmetadata', handleLoadedMetadata)
   715→    video.addEventListener('canplay', handleCanPlay)
   716→    return () => {
   717→      video.removeEventListener('loadedmetadata', handleLoadedMetadata)
   718→      video.removeEventListener('canplay', handleCanPlay)
   719→    }
   720→  }, [])
   721→
   722→  const formatDuration = (s: number) => {
   723→    if (!s || !isFinite(s)) return ''
   724→    const m = Math.floor(s / 60)
   725→    const sec = Math.floor(s % 60)
   726→    return ` • ${m}:${sec.toString().padStart(2, '0')}`
   727→  }
   728→
   729→  return (
   730→    <motion.div
   731→      initial={{ opacity: 0, y: -20 }}
   732→      animate={{ opacity: 1, y: 0 }}
   733→      transition={{ duration: 0.5 }}
   734→      className="mb-6 overflow-hidden rounded-3xl border-2 border-blue-400 bg-white/95 shadow-xl shadow-blue-500/20 backdrop-blur"
   735→    >
   736→      {/* Header */}
   737→      <div className="bg-gradient-to-r from-blue-600 via-sky-500 to-cyan-500 px-4 py-3 text-white sm:px-5">
   738→        <div className="flex items-center gap-2">
   739→          <div className="flex h-9 w-9 items-center justify-center rounded-xl bg-white/20 backdrop-blur">
   740→            <Play className="ml-0.5 h-5 w-5" />
   741→          </div>
   742→          <div className="flex-1 min-w-0">
   743→            <div className="text-sm font-black uppercase tracking-tight sm:text-base">
   744→              {title}
   745→            </div>
   746→            {subtitle && (
   747→              <div className="text-[10px] font-medium text-white/80 sm:text-xs">
   748→                {subtitle}{formatDuration(duration)}
   749→              </div>
   750→            )}
   751→          </div>
   752→          {isLoading && (
   753→            <div className="flex items-center gap-1 text-[10px] font-bold uppercase tracking-wide text-white/80">
   754→              <div className="h-3 w-3 animate-spin rounded-full border-2 border-white/30 border-t-white" />
   755→              загрузка
   756→            </div>
   757→          )}
   758→        </div>
   759→      </div>
   760→
   761→      {/* Video element with native controls */}
   762→      <div className="relative bg-black">
   763→        <video
   764→          ref={videoRef}
   765→          src={src}
   766→          controls
   767→          preload="metadata"
   768→          playsInline
   769→          className="block h-auto w-full"
   770→          style={{ aspectRatio: '16 / 9' }}
   771→        />
   772→      </div>
   773→
   774→      {/* Footer hint */}
   775→      <div className="flex items-center justify-between gap-2 border-t border-blue-100 bg-blue-50/40 px-4 py-2 text-[10px] font-bold uppercase tracking-wide text-blue-700/70 sm:text-xs">
   776→        <div className="flex items-center gap-1">
   777→          <BookOpen className="h-3 w-3" />
   778→          <span className="hidden sm:inline">Смотрите видео, затем читайте гайд</span>
   779→          <span className="sm:hidden">Видеоурок</span>
   780→        </div>
   781→        <div className="text-blue-700/50">
   782→          ▶ Встроенные элементы управления
   783→        </div>
   784→      </div>
   785→    </motion.div>
   786→  )
