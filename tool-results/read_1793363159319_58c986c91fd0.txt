   701→   701→function VideoPlayer({ src, title, subtitle }: VideoPlayerProps) {
   702→   702→  const videoRef = useRef<HTMLVideoElement>(null)
   703→   703→  const [isLoading, setIsLoading] = useState(true)
   704→   704→  const [duration, setDuration] = useState(0)
   705→   705→
   706→   706→  useEffect(() => {
   707→   707→    const video = videoRef.current
   708→   708→    if (!video) return
   709→   709→    const handleLoadedMetadata = () => {
   710→   710→      setDuration(video.duration || 0)
   711→   711→      setIsLoading(false)
   712→   712→    }
   713→   713→    const handleCanPlay = () => setIsLoading(false)
   714→   714→    video.addEventListener('loadedmetadata', handleLoadedMetadata)
   715→   715→    video.addEventListener('canplay', handleCanPlay)
   716→   716→    return () => {
   717→   717→      video.removeEventListener('loadedmetadata', handleLoadedMetadata)
   718→   718→      video.removeEventListener('canplay', handleCanPlay)
   719→   719→    }
   720→   720→  }, [])
   721→   721→
   722→   722→  const formatDuration = (s: number) => {
   723→   723→    if (!s || !isFinite(s)) return ''
   724→   724→    const m = Math.floor(s / 60)
   725→   725→    const sec = Math.floor(s % 60)
   726→   726→    return ` • ${m}:${sec.toString().padStart(2, '0')}`
   727→   727→  }
   728→   728→
   729→   729→  return (
   730→   730→    <motion.div
   731→   731→      initial={{ opacity: 0, y: -20 }}
   732→   732→      animate={{ opacity: 1, y: 0 }}
   733→   733→      transition={{ duration: 0.5 }}

   736→   736→      {/* Header */}

   739→   739→          <div className="flex h-9 w-9 items-center justify-center rounded-xl bg-white/20 backdrop-blur">
   740→   740→            <Play className="ml-0.5 h-5 w-5" />
   741→   741→          </div>
   742→   742→          <div className="flex-1 min-w-0">
   743→   743→            <div className="text-sm font-black uppercase tracking-tight sm:text-base">
   744→   744→              {title}
   745→   745→            </div>
   746→   746→            {subtitle && (
   747→   747→              <div className="text-[10px] font-medium text-white/80 sm:text-xs">

   749→   749→              </div>
   750→   750→            )}
